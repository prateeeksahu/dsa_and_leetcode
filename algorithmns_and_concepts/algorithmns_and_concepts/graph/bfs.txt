BFS - Breath First Search

in this traversal we go layer by layer, so if tree looks like this

   1
  / \
 2   3
/ \ / \
4 5 6 7

this traversal will go like 

1 -> 2  -> 3 -> 4 -> 5 -> 6 -> 7

and if it is a graph like this

    1
   / \
  2   3
 / \ / \
4  5 6  7
   \ /
    8


this traversal will go like

1 -> 2  -> 3 -> 4 -> 5 -> 6 -> 7 -> 8

nothing much to explain here

class Solution {
    
    private void bfs(ArrayList<ArrayList<Integer>> adj, int u, boolean[] v, ArrayList<Integer> al){
        Queue<Integer> q = new LinkedList<>();
        
        q.offer(u);
        v[u] = true;
        al.add(u);
        
        while(!q.isEmpty()){
            u = q.poll();
            
            for(int i = 0; i<adj.get(u).size(); i++){
                if(!v[adj.get(u).get(i)]){
                    q.offer(adj.get(u).get(i));
                    v[adj.get(u).get(i)] = true;
                    al.add(adj.get(u).get(i));
                }
            }
        }
    }
    // Function to return Breadth First Traversal of given graph.
    public ArrayList<Integer> bfsOfGraph(int V, ArrayList<ArrayList<Integer>> adj) {
        // Code here
        boolean[] v = new boolean[V];
        ArrayList<Integer> al = new ArrayList<>();
        
        bfs(adj, 0, v, al);
        
        return al;
    }
}
